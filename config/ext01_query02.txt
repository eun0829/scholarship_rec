-- ext01_query02.txt : 외부01 장학공지 '첨부파일' 목록 조회 (홈페이지 DB)
--
-- [수정 규칙]
--  * SELECT 에 반드시 있어야 하는 컬럼(이름 또는 별칭): BBS_ATCHMNFL_SEQ(첨부번호), BBS_ARTCL_SEQ(본문 글번호), ORGINL_NM(원본 파일명)
--    (있으면 사용: SORT_NO = 첨부 순서). 컬럼 순서는 상관없습니다.
--  * 다운로드 주소는 프로그램이 만듭니다: {EXT01_DOWN_BASE}/bbs/tukorea/{게시판}/{BBS_ATCHMNFL_SEQ}/download.do
--  * 바인드 변수(:setup, :days)와 {DATE_COND} 의 의미는 ext01_query01.txt 와 같습니다.
--  * 첨부에 삭제 플래그 컬럼이 있으면 아래 WHERE 에 AND f.삭제컬럼 = 'N' 처럼 추가하세요.
--  * 서브쿼리의 공고 조건은 ext01_query01.txt 의 WHERE 와 같게 유지하세요.

SELECT    f.BBS_ATCHMNFL_SEQ,
          f.BBS_ARTCL_SEQ,
          f.ORGINL_NM,
          f.SORT_NO
FROM      HOME.APP_BBS_ATCHMNFL f
WHERE     f.BBS_ARTCL_SEQ IN (SELECT  a.BBS_ARTCL_SEQ
                              FROM    HOME.APP_BBS_ARTCL a
                              WHERE   a.BBS_SETUP_SEQ = :setup
                              AND     a.TRASH_YN = 'N'
                              AND     {DATE_COND})
ORDER BY  f.BBS_ARTCL_SEQ, f.SORT_NO
